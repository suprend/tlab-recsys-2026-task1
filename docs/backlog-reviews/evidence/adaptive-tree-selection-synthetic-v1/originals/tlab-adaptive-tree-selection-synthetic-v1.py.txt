"""One finite synthetic selector check. No database or native ML operations."""
import argparse
import hashlib
import json
import math
from pathlib import Path
import resource
import signal
import sys
import time
from types import ModuleType

DRIVER = Path('/tmp/tlab-adaptive-tree-selection-study-v1.py')
DRIVER_SHA = 'a7c2a630dc289b50d279553a5079fb92484ce562fbb721f5858e716a9295e261'
CHECKOUT = Path('/tmp/tlab-public018-clean-20261015-once')
PUBLIC_SHA = 'ad27d69662e85a647c34c5853b64213a0526bf509e84f4dd639a7e93acf02a85'


def main(result):
    start_cpu, start_wall = time.process_time(), time.monotonic()
    resource.setrlimit(resource.RLIMIT_CPU, (9, 9))
    signal.signal(signal.SIGALRM, lambda *_: (_ for _ in ()).throw(TimeoutError('Synthetic wall fence')))
    signal.alarm(18)
    def check():
        peak = resource.getrusage(resource.RUSAGE_SELF).ru_maxrss
        peak = peak if sys.platform == 'darwin' else peak * 1024
        if time.process_time() >= 9 or time.monotonic() - start_wall >= 18 or peak > 512 * 2**20:
            raise RuntimeError('Synthetic resource fence')
    def audit(event, args):
        if (event.startswith(('subprocess.', 'socket.', 'os.exec', 'os.spawn', 'os.posix_spawn'))
                or event in ('os.system', 'os.fork', 'os.forkpty', 'sqlite3.connect')):
            raise RuntimeError('Forbidden process/network/database operation: ' + event)
        if event == 'import' and str(args[0]).split('.')[0] == 'catboost':
            raise ImportError('Native ML import forbidden')
        if event == 'open' and isinstance(args[0], (str, bytes)):
            path = Path(args[0].decode() if isinstance(args[0], bytes) else args[0])
            if (set(path.parts) & {'artifacts', 'data'} or path.suffix.lower() in
                    {'.parquet', '.db', '.duckdb', '.cbm', '.npz', '.npy', '.csv', '.gz'}):
                raise RuntimeError('Real input/data open forbidden')
    sys.addaudithook(audit)
    checks, passed, error = [], False, None
    try:
        def load(name, path, expected):
            raw = path.read_bytes()
            assert hashlib.sha256(raw).hexdigest() == expected
            module = ModuleType(name)
            module.__file__, module.__package__ = str(path), name.rpartition('.')[0]
            sys.modules[name] = module
            exec(compile(raw, str(path), 'exec', dont_inherit=True), module.__dict__)
            return module
        package = ModuleType('scripts')
        package.__path__ = [str(CHECKOUT / 'scripts')]
        sys.modules['scripts'] = package
        r = load('scripts.reproduce_frozen_comparison', CHECKOUT / 'scripts/reproduce_frozen_comparison.py', PUBLIC_SHA)
        recipe, _, _ = r.validate_recipe(CHECKOUT / 'configs/reproduction/frozen-bw-oct2019.json', check)
        binding = r.SourceBinding(recipe, check)
        binding.load_cores()
        selector = load('adaptive_selector_under_test', DRIVER, DRIVER_SHA)
        import numpy as np
        from scripts import train_baselines as b
        from scripts.ranking_metrics import evaluate
        class FakeModel:
            tree_count_ = 3
            def __init__(self, stages):
                self.stages, self.calls = stages, []
            def staged_predict(self, x, **kwargs):
                self.calls.append((x.copy(), kwargs))
                yield from self.stages
        def expect_reject(operation):
            try:
                operation()
            except (ValueError, RuntimeError):
                return
            raise AssertionError('Expected rejection')
        q = [('hit', 10, None, frozenset({10, 999})), ('offC', 20, None, frozenset({888})),
             ('emptyC', 30, None, frozenset({777})), ('negative', 40, None, frozenset())]
        data = b.Prepared(q, np.array([0, 0, 1, 1, 3, 3], dtype=np.int32),
                          np.array([10, 20, 30, 40, 50, 60], dtype=np.int64),
                          np.arange(6, dtype=np.float32).reshape(6, 1),
                          np.array([1, 0, 0, 0, 0, 0], dtype=np.int8), ['literal'])
        stages = [np.array(s, dtype=np.float64) for s in ([0, 1, 0, 0], [1, 1, 0, 0], [2, 1, 0, 0])]
        model = FakeModel(stages)
        best, curve, projection = selector.select_best_tree(model, data, maximum=3, check=check)
        discount, ideal = 1 / math.log2(3), 1 + 1 / math.log2(3)
        expected = [discount / ideal / 3, 1 / ideal / 3, 1 / ideal / 3]
        assert best == 2 and np.allclose(curve, expected, rtol=0, atol=1e-15)
        assert projection == dict(fullY_positive_queries=3, candidate_rows=4)
        assert len(model.calls) == 1 and np.array_equal(model.calls[0][0], data.x[:4])
        assert model.calls[0][1] == dict(prediction_type='RawFormulaVal', ntree_start=0,
                                       ntree_end=3, eval_period=1, thread_count=2)
        ranked = b.rankings(data, np.array([1, 1, 0, 0, 500, -500], dtype=np.float64))
        assert ranked[0].ranked_items == (10, 20)
        full = evaluate(ranked, 20).conditional_macro_query.ndcg
        assert abs(full - expected[1]) < 1e-15
        wrong_subset = evaluate([ranked[0]], 20).conditional_macro_query.ndcg
        assert abs(wrong_subset - 3 * full) < 1e-15 and abs(full - 1 / 3) > .01
        checks.append('Hand-computed fullY IDCG, all3positive denominator, empty/offC and integer tie; full cohort parity')
        s = .2
        tie, _ = selector._choose_curve([s, s + .75e-12, s + 1.5e-12], float, maximum=3, check=check)
        assert tie == 2
        checks.append('Global maximum tie band defeats chained-incumbent tolerance')
        for bad in (stages[:2], stages + stages[:1], [np.full(4, np.nan), *stages[1:]]):
            expect_reject(lambda bad=bad: selector.select_best_tree(FakeModel(bad), data, maximum=3, check=check))
        no_positive = b.Prepared([q[3]], np.array([0], dtype=np.int32), np.array([50], dtype=np.int64),
                                 np.zeros((1, 1), dtype=np.float32), np.zeros(1, dtype=np.int8), ['literal'])
        empty_model = FakeModel(stages)
        expect_reject(lambda: selector.select_best_tree(empty_model, no_positive, maximum=3, check=check))
        assert not empty_model.calls
        checks.append('Reject missing/excess/nonfinite stages and undefined positive cohort')
        empty = b.Prepared(q[1:3], np.empty(0, dtype=np.int32), np.empty(0, dtype=np.int64),
                           np.empty((0, 1), dtype=np.float32), np.empty(0, dtype=np.int8), ['literal'])
        empty_model = FakeModel([])
        best, curve, projection = selector.select_best_tree(empty_model, empty, maximum=3, check=check)
        assert best == 1 and curve == [0., 0., 0.] and not empty_model.calls
        assert projection == dict(fullY_positive_queries=2, candidate_rows=0)
        checks.append('All positive emptyC: exactzero curve and earliest tree without native empty call')
        binding.check()
        check()
        passed = True
    except BaseException as primary:
        error = dict(type=type(primary).__name__, message=str(primary))
        raise
    finally:
        peak = resource.getrusage(resource.RUSAGE_SELF).ru_maxrss
        body = dict(passed=passed, checks=checks, error=error, driver_sha256=DRIVER_SHA,
                    cpu_body=time.process_time()-start_cpu, wall_body=time.monotonic()-start_wall,
                    peak_rss_bytes=peak if sys.platform=='darwin' else peak*1024,
                    accounting='Process peak and main body only; Root external timer closes startup/exit',
                    actual_ML_operations=0, databases_opened=0, child_creation_attempts_forbidden=True)
        with result.open('x') as stream:
            json.dump(body, stream, sort_keys=True, indent=2, allow_nan=False)
            stream.write('\n')


if __name__ == '__main__':
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--result', type=Path, required=True)
    main(parser.parse_args().result)
