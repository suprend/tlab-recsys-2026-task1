def need(condition, message):
    if not condition:
        raise ValueError(message)


def check():
    if time.process_time() >= 8 or time.monotonic() - START >= 25:
        raise TimeoutError("Ordinary builder cooperative CPU/wall exhausted")
    rss = resource.getrusage(resource.RUSAGE_SELF).ru_maxrss
    if int(rss if sys.platform == "darwin" else rss * 1024) > 512 * 2**20:
        raise MemoryError("Ordinary builder cooperative SELF peak RSS exceeded")


def alarm(_number, _frame):
    raise TimeoutError("Builder alarm reached; external caller owns hard wall/reap")


def audit(event, args):
    if event in DENIED_EVENTS:
        raise RuntimeError("Denied ordinary-builder operation: " + event)
    if event == "import" and str(args[0]).split(".")[0] in FORBIDDEN_MODEL_ROOTS:
        raise RuntimeError("Model backend prohibited in no-fit fixture builder")


def file_ref(path):
    path = Path(path)
    check()
    need(not path.is_symlink() and path.is_file(), "Regular file required: " + str(path))
    size = path.stat().st_size
    need(size <= 32 * 2**20, "Fixture file exceeded finite 32MiB cap")
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        while block := stream.read(256 * 1024):
            check()
            digest.update(block)
    check()
    return {"path": str(path), "sha256": digest.hexdigest(), "bytes": size,
            "mode": format(stat.S_IMODE(path.stat().st_mode), "04o")}


def write_new(path, raw):
    path = Path(path)
    path.relative_to(PROJECT)
    check()
    path.parent.mkdir(parents=True, exist_ok=True)
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
    with os.fdopen(fd, "wb") as stream:
        stream.write(raw)
        stream.flush()
        os.fsync(stream.fileno())
    return file_ref(path)


def write_json(name, value):
    raw = (json.dumps(value, sort_keys=True, separators=(",", ":"), allow_nan=False) + "\n").encode()
    return write_new(PROJECT / name, raw)


def recheck_sources():
    for name, expected in {**COPY_PINS, "scripts/source_import.py": IMPORTER_PIN}.items():

    for name, expected in COPY_PINS.items():
        need(file_ref(PROJECT / name)["sha256"] == expected, "Exact copied input changed: " + name)


def load_collection_module():
    need(not any(name == "scripts" or name.startswith("scripts.") for name in sys.modules),
         "Fresh project namespace required")
    package = ModuleType("scripts")
    package.__path__ = [str(PROJECT / "scripts")]
    package.__package__ = "scripts"
    sys.modules["scripts"] = package
    module = importlib.import_module("scripts.development_collection")
    check()
    actual = {name for name in sys.modules if name.startswith("scripts.")}
    need(actual == IMPORTED_PROJECT, "Unexpected imported project module")
    for name in sorted(actual):
        path = Path(sys.modules[name].__file__).resolve()
        relative = "scripts/" + name.rsplit(".", 1)[1] + ".py"
        need(path == PROJECT / relative and file_ref(path)["sha256"] == COPY_PINS[relative],
             "Project module resolution differs: " + name)
    need(module.ROOT.resolve() == PROJECT, "Actual copied module ROOT differs")
    return module

def create_source():
    import duckdb
    need(duckdb.__version__ == "1.5.6", "Locked DuckDB version required")
    db = PROJECT / DB_NAME
    db.parent.mkdir(parents=True, exist_ok=False)
    con = duckdb.connect(str(db), config={"threads": "1", "memory_limit": "64MB",
                         "max_temp_directory_size": "0B", "enable_external_access": "false"})
    try:
        check()
        con.execute("CREATE TABLE events (" + ",".join(n + " " + t for n,t in SCHEMA) + ")")
        con.executemany("INSERT INTO events VALUES (?,?,?,?,?,?,?,?,?)", EVENTS)
        con.execute("CREATE TABLE ingest_metadata(note VARCHAR)")
        con.execute("INSERT INTO ingest_metadata VALUES (?)", [NOTE])
        need([tuple(r[:2]) for r in con.execute("DESCRIBE events").fetchall()] == SCHEMA, "Typed9 differs")
        need(con.execute("SELECT * FROM events ORDER BY event_time,user_id,product_id NULLS FIRST").fetchall() == EVENTS,
             "Literal typed9 roundtrip differs")
        need(con.execute("SELECT note FROM ingest_metadata").fetchall() == [(NOTE,)], "Fixture metadata differs")
        con.execute("CHECKPOINT")
        check()
    finally:
        con.close()
    check()
    return file_ref(db)  # Only after fixture metadata, checkpoint and close.
